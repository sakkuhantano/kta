CREATE PROCEDURE sp_KTA_selectLPC 
(
	@CompanyID VARCHAR(10),
	@LocationID VARCHAR(10),
	@UnitNo VARCHAR(10),
	@Period VARCHAR(6)
)

AS
/*
DECLARE @CompanyID VARCHAR(10)
SET @CompanyID='N00042032'
DECLARE @LocationID VARCHAR(10)
SET @LocationID='L01'
DECLARE @UnitNo VARCHAR(10)
SET @UnitNo='0526G'
DECLARE @Period VARCHAR(6)
SET @Period='202512'
*/
DECLARE @PrevPeriod VARCHAR(6)
SET @PrevPeriod=convert(varchar(6),cast(@Period+'01' AS SMALLDATETIME)-1,112)

SELECT h.KeyHD, h.DocumentNo, h.DocumentDate, h.DueDate, h.IncomeCode, h.CurrencyID, h.UnitNo, h.OwnerID, 
BillDate=ISNULL(z.BillDate,h.DocumentDate), d.RecordType,
d.RefNo, d.RKDate, d.Amount,
StartDate=cast(@PrevPeriod+RIGHT('00'+cast(
					CASE WHEN @Period<=convert(varchar(6),z1.StartingNewBilling,112) then z1.PreviousBillDate else z1.BillDate END 
			   AS VARCHAR(2)),2) AS SMALLDATETIME),
EndDate=cast(@Period+ RIGHT('00'+cast(z1.BillDate AS VARCHAR(2)),2) AS SMALLDATETIME)-1,
ISNULL(dd.Amount,0) OffsetAmount
from kta_t_h_outstanding h inner join kta_t_d_outstanding d on d.keyhd=h.keyhd 
inner join (kta_m_owner x inner join kta_m_unit y on x.companyid=y.companyid and x.locationid=y.locationid and x.ownerid=y.ownerid) 
on y.companyid=h.companyid and y.locationid=h.locationid and y.unitno=h.unitno
INNER JOIN kta_m_penalty z1 ON z1.CompanyID=h.CompanyID AND z1.LocationID=h.LocationID
LEFT JOIN kta_t_d_outstanding dd JOIN KTA_T_H_Outstanding hh ON dd.KeyHD=hh.KeyHD AND dd.RecordType=1 AND dd.Amount<0 ON hh.DocumentNo=h.DocumentNo AND hh.TxType=h.TxType AND hh.UnitNo=h.UnitNo
left join 
(
	SELECT h.CompanyID, h.LocationID, h.UnitNo, h.BillDate, d.InvoiceNo, d.IncomeCode 
	FROM kta_t_d_billing d, kta_t_h_billing h WHERE d.KeyHD=h.KeyHD AND d.IncomeCode IN ('WOL','WOM')
) z on z.companyid=h.companyid and z.locationid=h.locationid and z.unitno=h.unitno and z.incomecode=h.incomecode and z.invoiceno=h.documentno 
where x.penalty='y' and exists 
(
	select a.companyid,a.locationid,a.documentno,a.txtype 
	from kta_t_h_outstanding a inner join kta_t_d_outstanding b on a.keyhd=b.keyhd 
	where b.refno<>''
	and a.companyid=h.companyid and a.locationid=h.locationid and a.documentno=h.documentno and a.txtype=h.txtype and a.ownerid=h.ownerid  
	and duedate<=cast(@Period+RIGHT('00'+cast(z1.BillDate AS VARCHAR(2)),2) AS SMALLDATETIME)-1
	AND case when recordtype=1 then refdate ELSE rkdate END <=duedate
	AND case when recordtype=1 then refdate ELSE rkdate END<=cast(@Period+RIGHT('00'+cast(z1.BillDate AS VARCHAR(2)),2) AS SMALLDATETIME)-1
	group by a.companyid,a.locationid,a.documentno,a.txtype
	having sum(case when b.recordtype=1 then b.amount else -b.amount end)>0 
) 
and h.unitno=@UnitNo and (h.paiddate='' or convert(varchar(6),h.paiddate,112)=@PrevPeriod) 
and h.companyid=@CompanyID and h.locationid=@LocationID 
and d.refno<>'' 
order by d.keyhd, case when d.recordtype=1 then 1 when d.recordtype=3 then 2 else 3 end, d.rkdate;
GO

CREATE VIEW vw_KTA_T_H_Outstanding  
AS  
select h.*,  
Period=z2.ActivePeriod,  
PrevPeriod=DATEADD(MONTH,-1, z2.ActivePeriod+1)-1,  
BillDate=ISNULL(z1.BillDate,h.DocumentDate),  
StartDate=cast(convert(varchar(6),DATEADD(MONTH,-1, z2.ActivePeriod+1)-1,112)+  
RIGHT('00'+cast(CASE WHEN convert(varchar(6),z2.ActivePeriod,112)<=convert(varchar(6),z.StartingNewBilling,112) then z.PreviousBillDate else z.BillDate END  
AS VARCHAR(2)),2) AS SMALLDATETIME),  
EndDate=cast(CONVERT(VARCHAR(6),z2.ActivePeriod,112)+ RIGHT('00'+cast(z.BillDate AS VARCHAR(2)),2) AS SMALLDATETIME)-1  
from kta_t_h_outstanding h INNER JOIN kta_m_penalty z ON z.CompanyID=h.CompanyID AND z.LocationID=h.LocationID  
INNER JOIN m_active_period z2 ON z2.AutoNum='KCAR'+h.CompanyID+h.LocationID  
LEFT JOIN  
(  
SELECT h.CompanyID, h.LocationID, h.UnitNo, h.BillDate, d.InvoiceNo, d.IncomeCode  
FROM kta_t_d_billing d, kta_t_h_billing h WHERE d.KeyHD=h.KeyHD AND d.IncomeCode IN ('WOL','WOM','WOH')  
) z1 ON z1.CompanyID=h.CompanyID AND z1.LocationID=h.LocationID AND z1.UnitNo=h.UnitNo AND z1.IncomeCode=h.IncomeCode and z1.invoiceno=h.documentno  
WHERE locationno='';
GO
